<script>

    // Data Object - Like W3Schools Examples

            { name: "NFT Latte", price: "KSh 550", description: "Latte art minted as NFT", emoji: "🎨", type: "web3" },
            { name: "Smart Contract Cappuccino", price: "KSh 500", description: "Auto-loyalty points on chain", emoji: "📜", type: "web3" },
            { name: "Crypto Cold Brew", price: "KSh 480", description: "Pay with M-Pesa or Crypto", emoji: "💰", type: "web3" },
            { name: "Genesis Pour Over", price: "KSh 600", description: "Single origin, traceable on blockchain", emoji: "🌟", type: "web3" }

        web3Features: {
            loyalty: "Earn 'Bean Tokens' for every purchase",
            nftRewards: "Collect 5 NFT receipts, get 1 free coffee",

    // Global Variables - W3Schools Style
    let userPoints = 0;
    let nftCount = 0;
    const chatBox = document.getElementById("chatBox");
    const userInput = document.getElementById("userInput");
    const typing = document.getElementById("typing");
    const quickButtons = document.getElementById("quickButtons");

    // Initialize Chat - Like W3Schools onload
    window.onload = function() {
        setTimeout(() => {
            addBotMessage("Jambo! I'm Dennis, your Web3 Barista. ☕✨");
            setTimeout(() => {
                addBotMessage("I don't just make coffee—I create blockchain-verified experiences. Every cup earns you 'Bean Tokens' redeemable for NFTs!");
                showButtons(["View Web3 Menu", "Meet Dennis", "How does loyalty work?", "Pay with Crypto?"]);
            }, 800);
        }, 500);
    };

    // Function to Add Bot Message - W3Schools Function Style
    function addBotMessage(text, isHTML = false) {
        let msgDiv = document.createElement("div");
        msgDiv.className = "msg bot";
        
        let content = isHTML ? text : `<div class="bubble">${text}</div>`;
        msgDiv.innerHTML = `<div class="avatar">☕</div>${content}`;
        
        chatBox.appendChild(msgDiv);
        scrollToBottom();
    }

    // Function to Add User Message
    function addUserMessage(text) {
        let msgDiv = document.createElement("div");
        msgDiv.className = "msg user";
        msgDiv.innerHTML = `<div class="avatar">😊</div><div class="bubble">${text}</div>`;
        chatBox.appendChild(msgDiv);
        scrollToBottom();
    }

    // Function to Show Typing
    function showTyping() {
        typing.classList.add("active");
        scrollToBottom();
    }

    // Function to Hide Typing
    function hideTyping() {
        typing.classList.remove("active");
    }

    // Function to Scroll Chat
    function scrollToBottom() {
        chatBox.scrollTop = chatBox.scrollHeight;
    }

    // Function to Create Buttons - W3Schools Event Handling
    function showButtons(options) {
        quickButtons.innerHTML = ""; // Clear existing
        
        options.forEach(option => {
            let btn = document.createElement("button");
            btn.className = option.includes("Web3") || option.includes("Crypto") || option.includes("NFT") || option.includes("loyalty") ? "w3-button web3" : "w3-button";
            btn.innerHTML = option;
            
            // W3Schools Style Event Listener
            btn.onclick = function() {
                userInput.value = option;
                sendMessage();
            };
            
            quickButtons.appendChild(btn);
        });
    }

    // Main Send Function - W3Schools Style Logic
    function sendMessage() {
        let text = userInput.value.trim();
        if (text === "") return; // Validation like W3Schools
        
        addUserMessage(text);
        userInput.value = "";
        quickButtons.innerHTML = ""; // Clear buttons
        
        // Show typing effect
        showTyping();
        
        // Process response with delay (simulating thinking)
        setTimeout(function() {
            hideTyping();
            processMessage(text.toLowerCase());
        }, 1000);
    }

    // Message Processing - W3Schools If-Else Logic
    function processMessage(input) {
        let response = "";
        let isHTML = false;
        let nextButtons = [];

        // Menu Requests
        if (input.includes("menu") || input.includes("drink") || input.includes("coffee")) {
            response = createMenuHTML();
            isHTML = true;
            nextButtons = ["What is Blockchain Brew?", "Tell me about NFT Latte", "How do I pay?", "Meet Dennis"];
        }
        
        // Web3/Blockchain Questions
        else if (input.includes("web3") || input.includes("blockchain") || input.includes("loyalty") || input.includes("token")) {
            response = createWeb3InfoHTML();
            isHTML = true;
            nextButtons = ["View Web3 Menu", "Connect Wallet (Demo)", "Meet Dennis", "Location?"];
            userPoints += 10; // Earn points for learning
        }
        
        // Dennis Info
        else if (input.includes("dennis") || input.includes("barista") || input.includes("who")) {
            response = createDennisHTML();
            isHTML = true;
            nextButtons = ["View Web3 Menu", "How does loyalty work?", "Pay with Crypto?"];
        }
        
        // Specific Drinks
        else if (input.includes("blockchain brew")) {
            response = "Our <strong>Blockchain Brew</strong> is cold coffee steeped for 12 hours, served with a unique QR code that mints your experience as an NFT on Polygon blockchain. Collect them all! ⛓️☕";
            nextButtons = ["How much is it?", "What is NFT Latte?", "View full menu"];
        }
        
        else if (input.includes("nft latte")) {
            response = "The <strong>NFT Latte</strong> is where art meets blockchain. I create latte art, photograph it, and mint it as an NFT sent to your wallet. You own the art forever! Plus it tastes amazing. 🎨☕";
            nextButtons = ["Price?", "Other Web3 drinks?", "Meet Dennis"];
        }
        
        // Pricing
        else if (input.includes("price") || input.includes("cost") || input.includes("how much") || input.includes("ksh")) {
            response = `Our Web3 coffees range from <strong>${shopData.menu[0].price}</strong> to <strong>${shopData.menu[5].price}</strong>. Every purchase earns you 10 Bean Tokens. Collect 100 tokens = 1 free NFT coffee! 💎`;
            nextButtons = ["View Web3 Menu", "How does loyalty work?", "Pay with Crypto?"];
        }
        
        // Crypto Payment
        else if (input.includes("crypto") || input.includes("bitcoin") || input.includes("ethereum") || input.includes("pay") || input.includes("m-pesa")) {
            response = "We accept: <strong>M-Pesa</strong> (instant), <strong>Bitcoin</strong>, <strong>Ethereum</strong>, or <strong>USDC</strong>. Crypto payments get 5% bonus loyalty points! In Nairobi, we bridge traditional mobile money with Web3. 💰📱";
            nextButtons = ["View Web3 Menu", "How does loyalty work?", "Location?"];
        }
        
        // Location
        else if (input.includes("location") || input.includes("where") || input.includes("nairobi") || input.includes("find")) {
            response = `📍 <strong>${shopData.location}</strong><br><br>We're in the heart of Nairobi's tech district. Look for the coffee shop with the Ethereum logo! Perfect for remote work with reliable WiFi.`;
            nextButtons = ["View Web3 Menu", "Meet Dennis", "How does loyalty work?"];
        }
        
        // Wallet Connect (Demo)
        else if (input.includes("wallet") || input.includes("connect")) {
            response = createWalletHTML();
            isHTML = true;
            nextButtons = ["View Web3 Menu", "Check My Points", "Meet Dennis"];
            nftCount++; // Simulate earning NFT
        }
        
        // Check Points
        else if (input.includes("points") || input.includes("balance")) {
            response = createPointsHTML();
            isHTML = true;
            nextButtons = ["View Web3 Menu", "Connect Wallet (Demo)", "Redeem Rewards"];
        }
        
        // Greetings
        else if (input.includes("hi") || input.includes("hello") || input.includes("jambo") || input.includes("hey")) {
            response = "Jambo! ☕ Ready to explore the future of coffee? I'm Dennis, and I'm here to make your coffee experience unforgettable and blockchain-verified!";
            nextButtons = ["View Web3 Menu", "Meet Dennis", "How does loyalty work?"];
        }
        
        // Default Response
        else {
            response = "I'm learning every day! I can tell you about our <strong>Web3 coffee menu</strong>, how our <strong>blockchain loyalty</strong> works, or introduce myself. What interests you?";
            nextButtons = ["View Web3 Menu", "Meet Dennis", "How does loyalty work?"];
        }

        addBotMessage(response, isHTML);
        if (nextButtons.length > 0) {
            showButtons(nextButtons);
        }
    }

    // HTML Generator Functions - Like W3Schools InnerHTML Examples
    
    function createMenuHTML() {
        let html = "<div style='margin-bottom:15px'><strong>☕ Web3 Coffee Menu:</strong></div><div class='w3-row'>";
        
        shopData.menu.forEach(item => {
            html += `
                <div class="w3-col" onclick="selectDrink('${item.name}')">
                    <div style="font-size:2em">${item.emoji}</div>
                    <h4>${item.name}</h4>
                    <div class="price">${item.price}</div>
                    <p style="font-size:0.9em;color:#666">${item.description}</p>
                    ${item.type === 'web3' ? '<span style="color:#6366f1;font-size:0.8em">⛓️ Web3 Enabled</span>' : ''}
                </div>
            `;
        });
        
        html += "</div>";
        return html;
    }

    // Helper for drink selection

        userInput.value = "Tell me about " + drinkName;
        sendMessage();
    }

    userInput.addEventListener("keypress", function(event) {
        if (event.key === "Enter") {
            sendMessage();
        }
